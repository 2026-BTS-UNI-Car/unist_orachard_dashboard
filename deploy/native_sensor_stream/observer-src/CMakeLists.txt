cmake_minimum_required(VERSION 3.16)
project(ubts_private_actual_cdr_stream LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rmw REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(actual_cdr_stream actual_cdr_stream.cpp)
target_compile_features(actual_cdr_stream PRIVATE cxx_std_17)
target_compile_options(actual_cdr_stream PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(actual_cdr_stream rclcpp rmw sensor_msgs)
target_link_libraries(actual_cdr_stream OpenSSL::Crypto)
